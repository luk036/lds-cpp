45 template <typename T, typename Table>
46 constexpr auto vdc_digit_sum(unsigned long num, unsigned long base, const Table& weights)
47 -> T {
48 T res{};
49 std::size_t idx = 0;
50 while (num != 0) {
51 const auto remainder = num % base;
52 num /= base;
53 res += static_cast<T>(remainder) * weights[idx];
54 ++idx;
55 }
56 return res;
57 }

323 template <unsigned long Base = 2> constexpr auto vdc(unsigned long cnt) -> double {
324 auto reslt = 0.0;
325 auto denom = 1.0;
326 auto count = cnt;
327 while (count != 0) {
328 const auto remainder = count % Base;
329 count /= Base;
330 denom *= static_cast<double>(Base);
331 reslt += static_cast<double>(remainder) / denom;
332 }
333 return reslt;
334 }

138 template <unsigned long Base = 2> constexpr auto vdc(unsigned long cnt) -> double {
139 auto reslt = 0.0;
140 auto denom = 1.0;
141 auto count = cnt;
142 while (count != 0) {
143 const auto remainder = count % Base;
144 count /= Base;
145 denom *= double(Base);
146 reslt += double(remainder) / denom;
